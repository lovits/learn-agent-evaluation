"""Persist experiment records using names that cannot escape the results directory."""

import json
import os
import re
from pathlib import Path

from .knowledge import PROJECT

NAME = re.compile(r"(?:e0[12]|grade-e0[12]|e0[12]-suite)-[0-9a-f]{32}\.json")


def report_root():
    return Path(os.environ.get("RUNS_DIR", PROJECT / "runs"))


def report_path(name, folder=None):
    root = Path(folder or report_root()).resolve()
    target = (root / name).resolve()
    if not NAME.fullmatch(name) or not target.is_relative_to(root):
        raise ValueError("无效的实验文件名。")
    return target


def save(report, prefix="e01", folder=None):
    target = report_path(f"{prefix}-{report['run_id']}.json", folder)
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(
        json.dumps(report, ensure_ascii=False, indent=2, allow_nan=False) + "\n", encoding="utf-8"
    )
    return target

# 整套课程的正文标准

以Learn Claude Code固定版本448c75bb596917ad196cc1fb7558f459c5c50fdd的s01、s02等实际中文正文为范本，以已重写的第1课为本课程样例。每课从一个能执行的具体任务开始，每次只加入解决当前困难的代码。对照工具、权限、状态、错误和停止等功能契约，避免提前导入完整Agent。

## 正文顺序

只使用六个二级标题：问题、解决方案、工作原理、相对上一课的变更、试一下、接下来。第1课的第四部分改为相对一次模型问答的变更。工作原理用加粗的“第1步……”至“第N步……”组织约5至7个短步骤，依照真实数据流推进。每步先解释为什么需要这段代码，再给短片段，再讲输入输出和接入位置。完整代码单独链接。注明片段有省略，不能冒充可独立运行的入口。

主文约2000至3500中文字，按实际机制复杂度调整；代码不算字数目标。短句和平实带读为主，解释具体变量与文件，不用口号、机械排比、泛化感叹和每节重复小结。避免把测试数和请求数作为主线。

## 主文与附页

主文保留理解机制必需的公式与一个算例。完整公式、边界、SDK差异、prompt/schema/verdict/聚合/异步调用链放source_walkthrough.md；实验数值和错误修复放run_summary.md；练习与答案分开；验收放docs。移动细节要有明确链接，知识点不能消失。新增关键图须对应实际代码，必要时才画，不放图片占位。图中模型、Harness、Agent、裁判和验收器的职责必须清楚。

每课至少在工作原理中给出真实关键代码；正文应沿问题与数据流阅读，而不是指标名清单或验收报告。相对前课用简短表说明机制变化，试一下含完整命令、输出位置和2至3个可以在记录中找到答案的观察题。接下来承接下一课问题。

## 证据和来源

DeepEval/Ragas的方法和原生SDK实际调用分别说明，固定源码文件/符号/commit并链接官方文档。构造回答、手算、回放、历史真实运行和本轮实测分开。本轮文档改写默认保留原运行代码和固定示例，不把原实验说成重测；若修改行为必须另实测并留原记录。不要填写未知模型ID/费用。

# 综合项目：一条请求怎样走完

从第1课到第15课，我们分别观察循环、检索、评分、任务与运行状态。综合项目把这些能力接到同一个编程 Agent 上。代码仍由模型决定怎样修改；讲师脚本负责安排实验阶段，并把审批和提交操作记为讲师行为。

![执行与验收关系](../lessons/e16_regression/images/integrated-runtime.png)

图对应当前接线，重点展示执行、证据与验收的关系。[可编辑图](../lessons/e16_regression/images/integrated-runtime.drawio)单独保留。主循环内的 Goal 判断和压缩恢复在下面展开。

## 从哪里读起

先看 [runtime.py](runtime.py) 的 `Runtime.__init__`。它把固定版本的 s15 集成模块加载到指定工作区，让 s15 继续使用原生 s09 记忆；随后接入 s16 workflow 和 s17 GoalController。临时环境设置在加载后恢复，模型调用经过本机桥，知识工具读取挂载目录。四份上游代码没有改写，来源与哈希在 [provenance.json](../vendor/learn_claude_code/provenance.json)。

适配层增加了请求账本、工具观测、知识读取与批处理授权，以及工作目录绑定事务。最后一处来自实际复现的并发丢失更新，见 [能力契约与限制](capabilities.md)。因此，读上游时能找到原始行为，读适配层时能看清课程改了哪里。

## 一次普通任务

`Runtime.submit` 检查任务非空，取得 `agent_lock`，触发用户输入 hook，再把请求追加到 `history`。锁保护同一会话的消息修改；模型响应仍可能调用多个工具，也可能结束本轮。

`_goal_loop` 先调用原生循环。原生循环在每次请求前收取 cron、后台通知，运行上下文预算与压缩，再组装记忆、技能目录和当前工具池。`connect_mcp` 只改变已连接服务器；下一轮重新组装工具池后，模型才能调用新发现的具名工具。

`RecordedMessages.create` 保存实际发出的参数，再调用真正的 SDK。宿主 [integrated_bridge.py](../tools/integrated_bridge.py) 把 Messages 请求转换为模型可返回的 `answer` 与 `tool_calls`，容器随后执行工具。宿主模型没有进入练习目录执行 Bash；工具结果来自容器。

每个普通工具调用先经过 `PreToolUse`。拒绝时，拒绝原因直接成为相应 `tool_use_id` 的结果；允许时才分发 handler，触发 `PostToolUse`，把实际返回值写回消息。工具函数返回错误字符串时，trace 的函数调用状态仍可能为 `SUCCESS`，因为函数本身正常返回了。判断工具是否完成业务动作，还要读结果内容。

没有新工具调用后，原生循环触发停止 hook，并尝试提取长期记忆。公开 Goal 裁判随后只读取对话证据：未完成就带原因续轮；后台仍运行时延后判断；裁判异常、连续阻止达到上限或目标失败都有独立状态。`returned` 表示运行入口返回，最终业务结论由外部验收器给出。

## 任务、队友与工作目录

TodoWrite 是内存中的整张清单替换。Task 则保存在 `.tasks`，有稳定 ID、依赖和 owner。持久队友领取 Task 后，assignment 同时绑定任务 ID 和实际 cwd；文件操作使用这个目录。一次性 `task` 子 Agent 有自己的消息列表，返回摘要后结束；持久队友会进入 IDLE，等待消息或认领下一项就绪任务。

审批绑定当前任务与工作版本。队友提交计划后要等待当前请求的批准；旧任务的审批不能授权新任务。任务释放会重置 plan gate，因此后续任务仍需审批时，讲师或 Lead 必须重新提出要求。关机同样需要当前请求 ID 和正确发送方，普通消息不能伪造一次有效审批。

worktree 提供独立 working copy。组合实验中，队友在分支里写评审；讲师等待任务完成、IDLE 与关机确认后提交评审，再调用宿主侧移除函数。移除前检查任务、租约、后台 cwd 和未提交内容，移除后保留分支。它不限制进程在操作系统里的全部活动。

## 后台、调度与工作流

只有 `bash` 的 `run_in_background` 明确为布尔 `True` 时，才启动后台路径。主循环先收到占位结果，实际命令结束后收到一次完成或失败通知。进程组清理覆盖普通结束和 SIGTERM；另建 session 的进程可能脱离原组。

cron 调度器按真实时钟每秒检查。持久的一次性任务先写 `pending_delivery`，再入队，直到包含它的模型请求成功才确认删除。崩溃恢复可能再次交付，所以应按至少一次交付设计幂等动作。这个承诺没有扩展到“任意损坏文件都能恢复”。

workflow 的编排由可信脚本保存。`course-review` 并行检查 API 合同与测试覆盖，再把候选意见交给验证步骤。每个成功步骤按名称、prompt、schema 生成语义键并追加 journal；resume 读取同一个 run ID 的快照、参数和日志。命中缓存只表示复用先前结果，不会使错误意见变正确。实现细节和失败恢复见 [第16课源码导读](../lessons/e16_regression/source_walkthrough.md)。

## 证据怎样进入评测

[traces.py](traces.py) 在固定版本 DeepEval 完成 trace 的边界保存原生对象，再保留原来的结束动作。同步调用按真实 parent UUID 组织；新队友线程产生的独立根保留独立根，没有补造跨线程父子关系。请求账本另外保存每次发送、响应、错误和时间。

[第16课采样器](../lessons/e16_regression/code.py)通过原生响应 ID 找到修复轮的 trace，再保存工具名、实际参数、输出及源 span UUID。主 Agent 的文字、检索规范和测试输出各有来源。两种 Faithfulness 证据范围分别评分，避免把规范要求误当成操作已发生。

验收在另一个断网容器中读取工作区。业务四文件复制到受限检查环境，验证两个 API、签名、调用关系和原测试；原生任务、记忆索引、工作流日志、定时交付和 Git 工作目录另按状态合同检查。错误与越界任务还要把实际工具尝试、拒绝结果和文件保持情况连起来。模型的完成声明、团队摘要和框架高分都不会免除这些检查。

## 一个在并发实验中发现的问题

原生 `create_worktree` 先读取待领取 Task，再执行 Git，最后保存带 worktree 的 Task。它只持有进程内锁；另一个进程可以在 Git 期间领取任务并写入 owner。绑定方随后保存旧对象，把这次认领覆盖回 `pending`。

失败实验先在真实 `git worktree add` 前暂停绑定，让另一个 Python 进程完成认领，再继续绑定。原始结果确实丢失 owner，保留在失败日志中。课程适配器现在把整个绑定操作放进原生 `task_store_lock`，与认领使用同一把跨进程锁。修复后的同一个实验检查到锁已持有、owner 保留，领取者 cwd 指向绑定目录。代价是 Git 创建期间其他任务写操作等待这把锁，适用于本课程的小任务板。

## 保留的边界

原生 TodoWrite 接受空清单和多个进行中项目；第13课的严格教学实现与它不同。释放 assignment 后 plan gate 会回到 `not_required`，因此下一项任务要重新声明审批要求。

原生 workflow 的 `SimpleJsonSchema` 只实现一个小子集，`integer` 分支也接受浮点数；它不能当作完整 JSON Schema 校验器。原生 durable cron 加载器遇到损坏内容会吞掉异常，不能承诺损坏恢复。原生 GoalController 可以从事件恢复条件，当前 CLI 不提供跨进程恢复整个会话。

本机模型桥不强制供应商 `max_tokens`，也没有 logprobs，所以截断、429/529 和上下文超长分支使用明确的协议夹具。模型版本与费用未知，报告保留为空。MCP 使用上游的进程内 stand-in：docs 读取真实挂载文件，deploy 示例仍模拟。工作目录和进程组分别提供 working copy 隔离与生命周期清理，不构成通用的恶意代码隔离保证。

PROJECT = Path(__file__).resolve().parent.parent
UPSTREAM = PROJECT / "vendor/learn_claude_code"


def json_value(value):
    if hasattr(value, "model_dump"):
        return json_value(value.model_dump(mode="json"))
    if isinstance(value, dict):
        return {k: json_value(v) for k, v in value.items()}
    if isinstance(value, (list, tuple)):
        return [json_value(v) for v in value]
    return value


def load_module(lesson):
    name = "course_" + lesson + "_" + uuid.uuid4().hex
    spec = importlib.util.spec_from_file_location(name, UPSTREAM / lesson / "code.py")
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module


class RecordedMessages:
    def __init__(self, sdk, maximum=80):
        self.sdk, self.maximum = sdk, maximum
        self.calls = []
        self.lock = threading.Lock()
        self.messages = self
        self.ledger = None

    def _write_ledger(self, event):
        if self.ledger is not None:
            with self.ledger.open("a") as handle:
                handle.write(json.dumps(event, ensure_ascii=False) + "\n")

    @observe(name="integrated_model", type="llm")
    def create(self, **kwargs):
        request = json_value(kwargs)
        with self.lock:
            if len(self.calls) >= self.maximum:
                raise RuntimeError("Declared application/support request budget exhausted")
            row = {
                "request": copy.deepcopy(request),
                "thread": threading.current_thread().name,
                "started_at_monotonic": time.monotonic(),
                "response_received": False,
            }
            self.calls.append(row)
            index = len(self.calls) - 1
            self._write_ledger({"event": "request_started", "index": index, **row})
        try:
            result = self.sdk.messages.create(**request)
            with self.lock:
                row.update(response_received=True, response=json_value(result))
            update_current_span(input=request["messages"], output=row["response"])
            return result
        except Exception as error:
            with self.lock:
                row["error"] = {"type": type(error).__name__, "message": str(error)}
            raise
        finally:
            with self.lock:
                row["ended_at_monotonic"] = time.monotonic()
                self._write_ledger({"event": "request_ended", "index": index, **row})

    def submit(self, query, goal=None):
        if not isinstance(query, str) or not query.strip():
            raise ValueError("A nonempty task is required")
        with self.host.agent_lock:
            self.goal.begin_query()
            if goal is not None:
                self.goal.set_goal(goal)
            self.host.trigger_hooks("UserPromptSubmit", query)
            self.state["active_user_request"] = query
            self.history.append({"role": "user", "content": query})
            self.host.agent_loop(self.history, self.context, query)
        return self.snapshot()

    def snapshot(self):
        with self.model.lock:
            calls = copy.deepcopy(self.model.calls)
        with self.host.background_lock:
            backgrounds = copy.deepcopy(self.host.background_tasks)
        history = json_value(self.history)
        last = next((m["content"] for m in reversed(history) if m["role"] == "assistant"), "")
        final_text = (
            last
            if isinstance(last, str)
            else "\n".join(b.get("text", "") for b in last if b.get("type") == "text")
        )
        return json_value(
            {
                "workspace": str(self.workspace),
                "history": self.history,
                "final_text": final_text,
                "model_calls": calls,
                "audit": self.audit,
                "goal_events": self.goal_events,
                "goal_active": self.goal.active is not None,
                "goal_last_status": self.goal.last_status,
                "backgrounds": backgrounds,
                "tasks": [self.host.asdict(t) for t in self.host.list_tasks()],
                "active_teammates": dict(self.host.active_teammates),
                "todos": copy.deepcopy(self.host.CURRENT_TODOS),
                "protocol_requests": [
                    self.host.asdict(p) for p in self.host.pending_requests.values()
                ],
                "teammate_assignments": copy.deepcopy(self.host.teammate_assignments),
                "schedules": [self.host.asdict(j) for j in self.host.scheduled_jobs.values()],
            }
        )

    def wait_until_settled(self, seconds):
        deadline = time.monotonic() + seconds
        while True:
            with self.host.agent_lock:
                with self.host.cron_lock:
                    scheduled = bool(self.host.cron_queue) or any(
                        not j.recurring for j in self.host.scheduled_jobs.values()
                    )
                with self.host.team_lock:
                    working = any(s != "idle" for s in self.host.active_teammates.values())
                pending = (
                    scheduled
                    or working
                    or self._background_running()
                    or self.host.has_pending_background()
                    or self.host.BUS.peek("lead")
                )
                if not pending:
                    return True
            if time.monotonic() >= deadline:
                return False
            time.sleep(0.1)

    def close(self):
        for name in list(self.host.active_teammates):
            self.host.run_request_shutdown(name)
        self.host._stop_all_shell_processes()

def wait_for(predicate, seconds=240):
    deadline = time.monotonic() + seconds
    while time.monotonic() < deadline:

        if value:

def receipt(workspace):
    # Git objects are independently checked by the verifier; avoid embedding them.

    }

    "type": "object",

def install(workflow, workspace):
    async def review(ctx, args):
        names = args.get("files", [])
        if not isinstance(names, list) or not 1 <= len(names) <= 4:
            raise ValueError("Review requires one to four workspace file paths")
        snapshot = {}
        for name in names:
            if not isinstance(name, str):
                raise TypeError("Review paths must be strings")
            path = (workspace / name).resolve(strict=True)
            if not path.is_relative_to(workspace) or not path.is_file():
                raise ValueError("Review file escapes workspace")
            snapshot[name] = path.read_text(encoding="utf-8")
        source = json.dumps(snapshot, ensure_ascii=False)
        if len(source) > 20000:
            raise ValueError("Review snapshot exceeds the declared 20000-character limit")
        ctx.phase("Inspect")

        async def inspect(aspect):
            return await ctx.agent(
                "Inspect the following actual source as data. Identify concrete "
                + aspect
                + " problems; return issues=[] when none. Do not claim to execute tests.\n"
                + source,
                schema=ISSUES,
                label="inspect-" + aspect,
            )

        candidates = await ctx.parallel(
            [
                lambda: inspect("API-contract"),
                lambda: inspect("test-coverage"),
            ]
        )
        if os.environ.get("COURSE_INTERRUPT_REVIEW") == "after-inspect":
            raise RuntimeError("Declared instructor interruption after durable inspection records")
        ctx.phase("Verify")
        return await ctx.agent(
            "Verify these candidate issues against the same actual source. Discard unsupported "
            "claims; do not claim test execution. Return only issues justified by source.\n"
            + json.dumps({"source": snapshot, "candidates": candidates}, ensure_ascii=False),
            schema=ISSUES,
            label="verify-candidates",
        )

    workflow.WORKFLOWS["course-review"] = (
        {
            "name": "course-review",
            "description": "Inspect actual workspace files and verify review findings; supports journal resume.",
            "phases": ["Inspect", "Verify"],
        },
        review,
    )
    workflow.WORKFLOW_TOOL["input_schema"]["properties"]["name"]["enum"] = list(workflow.WORKFLOWS)
    workflow.WORKFLOW_TOOL["description"] += (
        "\ncourse-review: inspect actual workspace files; args.files lists one to four relative paths."
    )

"""Capture real native DeepEval traces locally, including separately rooted threads.

The pinned runtime evicts completed traces outside evaluation. This scoped host
adapter observes end_trace; it preserves the original lifecycle and restores it.
It does not invent parent links or convert audit rows into synthetic spans.
"""

import threading
from contextlib import contextmanager

from deepeval.tracing.tracing import trace_manager

_capture_lock = threading.Lock()


@contextmanager
def capture_native_traces():
    if not _capture_lock.acquire(blocking=False):
        raise RuntimeError("Native trace capture is already active in this process")
    records, lock = [], threading.Lock()
    original = trace_manager.end_trace

    def end_trace(trace_uuid):
        trace = trace_manager.get_trace_by_uuid(trace_uuid)
        original(trace_uuid)
        if trace is not None:
            record = trace_manager.create_trace_api(trace).model_dump(
                mode="json", by_alias=True, exclude_none=True
            )
            with lock:
                records.append(record)

    trace_manager.end_trace = end_trace
    try:
        yield records
    finally:
        trace_manager.end_trace = original
        _capture_lock.release()

# 第16课参考答案

1. Goal 是读对话的自动判断；workflow completed 是编排走完；独立验收读取实际文件、执行限定行为并检查状态。三者作用不同，前两者不能代替最终事实检查。
2. 实际 `toolSpans.input` 含工具名与参数，output 是返回值；通过 trace 中的 UUID/parent UUID 连接。同步工具处在真实主轮作用域，原生队友新线程可能自成根。重建不存在的父子关系会把分析者推断写成执行事实。
3. 一个匹配、一个缺失：参考覆盖为1/2，precision为1，recall为1/2，所以 F1为2/3。exact match要求整个投影一致；Ragas Accuracy还乘工具名序列对齐因子。分母和对齐方式不同，不能直接取平均。
4. 默认为(3+2)/5=1，惩罚后为3/5=0.6。原始裁判已把“测试成功”标成证据不足的 borderline；默认聚合仍计通过。规范描述要求，执行事实还需要实际工具与外部证据。
5. 记录应包含 `tool_name`、真实 `input_parameters`、`output`、源 UUID 和内容哈希。原始轨迹已经产生这些字段，采样器此前省掉了参数。重新投影和重新请求裁判都不是重新运行 Agent。
6. 只接受每个关键项严格为 `True`。权限失败和覆盖 `None` 都应列入 rejected，业务通过不能抵消。错误保留为错误，不能换成一个看似可平均的低分或默认通过。
7. 旧真实组件日志有两条检查记录；新进程读取同一 run ID 后两步 cached，只新增一次验证。语义键包含 label、prompt、schema，改变这些内容会改变缓存语义。缓存中的内容还要按 schema 验证；损坏日志应报错并保留。
8. 绑定读 pending → 绑定开始 Git → 另一进程认领并写 owner → Git结束 → 绑定保存旧对象 → owner丢失。修复要让验证、Git创建和最终保存共用认领使用的跨进程锁。持锁期间其他写任务操作需要等待。
9. 审批属于当前工作身份；旧 request ID、任务或工作版本与新 assignment 不符。原生完成释放会把 gate 回到 not_required；第二个任务仍需审批时，应在任务可被认领前重新要求计划，并让新请求完成审批。
10. 重新入队，直到携带该 prompt 的模型请求成功再确认。动作可按稳定业务键写唯一结果，已有相同结果则读回确认；模型接收成功与业务动作成功仍是两个不同边界。队列的至少一次不保证任意业务操作恰好一次。
11. 文件不存在也可能是模型根本没尝试、请求失败或写错路径。还需看到指定工具和参数的一次 PreToolUse、同 ID 的拒绝 tool_result、未进入 PostToolUse、原始文件哈希不变，以及无绕行产生的额外文件。
12. 合格协议能由别人按同一入口复核，区分模型执行、构造退化、夹具和历史记录。知识/代码/提示/数据与模型可知版本应保存，未知型号或费用留空。验收范围限定到自己的任务与状态，不把小样本通过写成一般可靠性证明。

可选实验只说明该次诊断中的变化。首轮结果已经被看到，后续修订不能包装成独立留出结果；这次仅规范证据的 Ragas 分数也发生小幅变化，正好提醒我们注意拆句和模型判断的波动。

"""Project one actual native repair trace into declared evaluation samples."""

import copy
import hashlib
import json


def sha(value):
    return hashlib.sha256(value.encode()).hexdigest()


def trace_samples(report):
    if report.get("status") != "returned" or report.get("error"):
        raise ValueError("Incomplete integration run cannot become an accepted regression sample")
    if report.get("origin") != "live_native_integration_with_declared_instructor_orchestration":
        raise ValueError("This projection requires the declared native integration experiment")
    worker = next(
        (
            c
            for c in report["model_calls"]
            if c["request"].get("tools")
            and c.get("response_received")
            and c["response"]["stop_reason"] == "end_turn"
        ),
        None,
    )
    if worker is None:
        raise ValueError("No completed worker response")
    message_id = worker["response"]["id"]
    matches = [
        trace
        for trace in report["native_traces"]
        if any(span.get("output", {}).get("id") == message_id for span in trace.get("llmSpans", []))
    ]
    if len(matches) != 1:
        raise ValueError("Worker response must link to exactly one actual native trace")
    trace = matches[0]
    tools = sorted(trace.get("toolSpans", []), key=lambda s: s["startTime"])
    response = "\n".join(b["text"] for b in worker["response"]["content"] if b["type"] == "text")
    task = worker["request"]["messages"][0]["content"]
    documents = []
    for span in tools:
        name = span["input"]["name"]
        text = json.dumps(
            {
                "tool_name": name,
                "input_parameters": span["input"]["args"],
                "output": span["output"],
            },
            ensure_ascii=False,
        )
        if len(text) > 20000:
            raise ValueError(
                "Over-limit tool evidence requires an explicit separate sampling policy"
            )
        documents.append(
            {
                "id": span["uuid"],
                "path": "native:" + span["uuid"],
                "text": text,
                "sha256": sha(text),
                "tool_name": name,
            }
        )
    retrieved = [d for d in documents if d["tool_name"] == "mcp__docs__search"]
    if not retrieved or not response:
        raise ValueError("Native repair trace lacks actual retrieval evidence or final text")
    common = {
        "question": task,
        "response": response,
        "origin": "actual_worker_response_projected_from_native_trace",
        "source_trace_uuid": trace["uuid"],
        "source_message_id": message_id,
        "human_reviewed": False,
    }
    faithful = [
        {
            **common,
            "id": "documents_only",
            "documents": retrieved,
            "scope": "Report support from retrieved specification only; does not prove test execution.",
        },
        {
            **common,
            "id": "observed_tools",
            "documents": documents,
            "scope": "Report support from this repair turn's observed tool outputs, including tests; not a pure RAG-only comparison.",
        },
    ]
    predicted = [
        {"name": span["input"]["name"], "args": span["input"]["args"], "source_span": span["uuid"]}
        for span in tools
        if span["input"]["name"] == "read_file"
        and span["input"]["args"].get("path") in {"shipping.py", "checkout.py"}
    ]
    baseline = {
        "id": "native_implementation_read_projection",
        "task": task,
        "origin": "actual_native_tool_spans_projected_to_implementation_reads",
        "source_trace_uuid": trace["uuid"],
        "predicted": predicted,
        "expected": [
            {"name": "read_file", "args": {"path": "shipping.py"}},
            {"name": "read_file", "args": {"path": "checkout.py"}},
        ],
        "scope": "Declared read-file sequence diagnostic, not a unique correct whole-task trajectory or completion criterion.",
    }
    degraded = copy.deepcopy(baseline)
    degraded.update(
        id="injected_missing_checkout_read",
        origin="instructor_deleted_a_projection_event_no_new_agent_execution",
    )
    degraded["predicted"] = [
        c for c in degraded["predicted"] if c["args"].get("path") != "checkout.py"
    ]
    return {
        "faithfulness": faithful,
        "tools": [baseline, degraded],
        "projection_version": "native-repair-v2-tool-input-and-output",
        "new_agent_requests": 0,
    }


def critical_gate(checks):
    if not isinstance(checks, dict) or not checks:
        raise ValueError("Named critical checks are required")
    rejected = [name for name, passed in checks.items() if passed is not True]
    return {
        "passed": not rejected,
        "rejected": rejected,
        "rule": "Every named critical check must be exactly True; missing/unscored checks fail closed.",
    }


def trace_digest(report):
    return sha(json.dumps(report["native_traces"], ensure_ascii=False, sort_keys=True))

# 第16课源码导读：从执行证据追到分数

本课继续使用 [固定版本](../../references/versions.json)，没有因官网示例更新而自动升级依赖。DeepEval 为 `34db63dd005ebc9a905c8ef4741ae0687c93c14c`，Ragas 为 `8c63f10d500c44efbf24fdb9e2920de2104c321c`。以下指出本课真正调用的入口；批量评价入口另外解释，避免把源码阅读写成执行记录。

## 1. 先找到实际对象

`Runtime.submit → _goal_loop → 原生 agent_loop → RecordedMessages.create` 形成一次主轮。模型响应由本机桥转换成原生 Message；具名工具由 `PreToolUse → call_tool_handler → PostToolUse` 执行，工具结果与对应调用 ID 一起回到消息。Goal 裁判读同一会话的证据，但最终文件与状态由另一个容器验收。

原生 trace 通过 [traces.py](../../integrated_agent/traces.py) 保存。固定版本结束 trace 后会在非评价模式下清除已完成对象；课程在结束边界记录 `create_trace_api(...).model_dump`，再恢复原方法。它是依赖固定版本内部接口的适配，升级时必须重测。

trace 的 `SUCCESS` 表示被观测函数正常返回。例如文件工具返回 `Error: ...` 字符串，函数本身没有抛异常，也可能得到 `SUCCESS`。文件是否存在、任务是否完成，应读返回内容并执行相应验收。

本轮还有一条观测失败记录：把整个 Runtime 当成被装饰函数的普通参数，框架序列化其公开对象图，进程在首个模型请求前以137结束。已移除讲师编排函数上的装饰，仅观测原生执行边界。原容器自动移除，未取得 OOMKilled 元数据，因此不能把137单独写成“已证明内存溢出”。原始协议、源码快照和终止记录都保留。

## 2. trace 转成样本时保留什么

[trace_samples](code.py)先找首个真正完成的工作响应，以 `response.id` 连接原生 LLM span，再定位唯一 trace。工具 span 按原生时间排序；采样保留工具名、实际参数、返回值、源 UUID 和内容哈希。它没有按模型的完成总结倒推一条理想轨迹。

首次投影只留下工具输出，测试 `OK` 仍在，产生它的命令却不在。裁判因此无法核实“运行的是哪条命令”。修订版把实际参数一起加入证据；原 Agent、业务文件和运行记录保持原样，投影与新评分另外冻结。这是查看首轮结果后的开发诊断，不能称作独立留出集上的提升。

两份 Faithfulness 样本共用同一个真实回答：第一份只给实际 docs 检索输出；第二份给修复轮的工具观测。后者含测试和编辑记录，证据范围更宽，不能把分数差异归成检索器质量变化。规范中的“必须保持原测试”也不等于“本次已经保持”。

工具样本只投影两个实现文件的 `read_file` 调用。参考为先读 shipping、后读 checkout 的声明序列；实际参数不做悄悄归一化。删除 checkout 读取的版本是讲师构造的退化投影，新增 Agent 请求为0。全任务可以有其他合理工具路径，这个分数只诊断指定投影。

## 3. DeepEval Faithfulness 的完整链

固定源码：[faithfulness.py](https://github.com/confident-ai/deepeval/blob/34db63dd005ebc9a905c8ef4741ae0687c93c14c/deepeval/metrics/faithfulness/faithfulness.py)、[QAG 聚合与 schema 适配](https://github.com/confident-ai/deepeval/blob/34db63dd005ebc9a905c8ef4741ae0687c93c14c/deepeval/metrics/utils/qag.py)、[输入校验](https://github.com/confident-ai/deepeval/blob/34db63dd005ebc9a905c8ef4741ae0687c93c14c/deepeval/metrics/utils/test_case.py)。

本课复用第5课 `evaluate`：构造 `LLMTestCase(input, actual_output, retrieval_context)`，调用原生 `FaithfulnessMetric.measure`，明确使用课程自定义裁判。`prepare_measure` 校验该指标所需字段，并重置状态与成本；对象能构造成功，不代表它具备某个指标的全部输入。

同步 llm 路径先从证据生成 truths，再从回答生成 claims，最后按 truths 判断每条 claim。每一步用模板形成 prompt，交给 `DeepJudge.generate(prompt, schema)`；桥的原始文字通过 `schema.model_validate_json` 变成结构化对象。verdict 仍是裁判的判断，schema 只保证结果形状。异步 `a_measure` 并行提取 truths 与 claims，再生成 verdict 并聚合；共享可变 metric 实例不应同时测不同样本。

默认通过集合为 `YES` 与 `BORDERLINE`，开启 `penalize_ambiguous_claims` 后只通过 `YES`。对同一组 verdict 重算：

```text
默认分数 = (YES 数 + BORDERLINE 数) / 有效 verdict 数
惩罚含糊项 = YES 数 / 有效 verdict 数
```

没有有效 verdict 时，固定 QAG 默认返回1；严格模式另按阈值钳制低分。因此需要保存声明数、verdict 和异常，不能把空声明高分解释为高质量答案。本课 `include_reason=False` 省去整体验因请求，仍保存逐声明理由与 verdict。

官方 [Faithfulness 文档](https://deepeval.com/docs/metrics-faithfulness)说明用途；判分细节与空输入边界以固定源码及本课记录为准。

## 4. Ragas Faithfulness 的完整链

固定源码：[collections/faithfulness/metric.py](https://github.com/vibrantlabsai/ragas/blob/8c63f10d500c44efbf24fdb9e2920de2104c321c/src/ragas/metrics/collections/faithfulness/metric.py)、[prompt 与输出模型](https://github.com/vibrantlabsai/ragas/blob/8c63f10d500c44efbf24fdb9e2920de2104c321c/src/ragas/metrics/collections/faithfulness/util.py)。

本课使用 `metrics.collections.Faithfulness.ascore(user_input, response, retrieved_contexts)`。入口拒绝缺失的回答、问题和上下文。`_create_statements` 将问题、回答送入 `StatementGeneratorPrompt`，按 `StatementGeneratorOutput` 取得原子声明；`_create_verdicts` 将上下文与声明送入 NLI prompt，按 `NLIStatementOutput` 取得每项0/1。最后返回 `MetricResult.value`：

```text
score = 明确获支持的 statement 数 / statement 总数
```

无声明或空 verdict 集合可返回 NaN。本课将其记录为 `undefined_nan`、`score=None`；网络和解析错误另记 `error`。`CapturingJudge` 保存 schema 名及实际结构化结果，能查到每条分数从哪里来。

与 DeepEval 的默认值不同，Ragas 没有把“含糊但未被反驳”统一计为支持。首次实测还把一句报告拆成了不同数量的声明。即使使用同一个后端，prompt、拆句和聚合不同，也不会天然得到相同分数。官方定义见 [Ragas Faithfulness](https://docs.ragas.io/en/stable/concepts/metrics/available_metrics/faithfulness/)。

## 5. 工具指标为什么给出不同分数

本课复用第10课 [score_rules](../e10_tool_evaluation/metrics.py)，实际调用两个框架的原生实现，裁判请求为0。

DeepEval 的 `LLMTestCase.tools_called/expected_tools` 持有 `ToolCall`。只给预期工具时，`ToolCorrectnessMetric` 可做确定性比较；课程分别检查工具名、参数、顺序和 exact match。按两个预期调用计算，少一个可得到0.5，exact match则为0。若改为 `available_tools` 语义判断模式，算法与是否请求裁判都会变化，不能沿用这组规则结论。

Ragas collections `ToolCallAccuracy` 从 `AIMessage.tool_calls` 取预测，与独立提供的 `reference_tool_calls` 比较。先判断工具名序列是否对齐，再比较配对参数；额外或缺失调用可让序列因子为0。flexible 允许换序，仍要求名字多重集对齐。`ToolCallF1` 另按集合匹配计算 precision/recall，因此少一个时可得到约0.6667；重复去重也是它的边界。

固定源码：[DeepEval ToolCorrectness](https://github.com/confident-ai/deepeval/blob/34db63dd005ebc9a905c8ef4741ae0687c93c14c/deepeval/metrics/tool_correctness/tool_correctness.py)、[Ragas ToolCallAccuracy](https://github.com/vibrantlabsai/ragas/blob/8c63f10d500c44efbf24fdb9e2920de2104c321c/src/ragas/metrics/collections/tool_call_accuracy/metric.py)。

## 6. 批量评价入口与 CI 的职责

DeepEval [evaluate.py](https://github.com/confident-ai/deepeval/blob/34db63dd005ebc9a905c8ef4741ae0687c93c14c/deepeval/evaluate/evaluate.py)先校验测试与指标配置，再按 async/cache/error/display 配置执行，整理逐测试 `EvaluationResult`。本课为保存每一阶段的原始响应，直接调用 metric；没有把阅读批量入口当成它已被本课运行。

Ragas 固定版本 [evaluation.py](https://github.com/vibrantlabsai/ragas/blob/8c63f10d500c44efbf24fdb9e2920de2104c321c/src/ragas/evaluation.py)的旧批量 `evaluate` 包装 `aevaluate`，与本课 collections `ascore` 接口分开。官方 [CI 指南](https://docs.ragas.io/en/stable/howtos/applications/add_to_ci/)给回归思路，旧示例参数不能直接拼接到当前签名。DeepEval 的 [CI 指南](https://deepeval.com/guides/guides-regression-testing-in-cicd)同样需要按本课固定环境执行。

CI 运行断网的可重复合同测试，检查已保存 bad case 与门禁；真实 Agent 和裁判实验通过独立命令运行并留证。没有设定自动夜间任务，避免把课程里的发布策略解释成已部署的调度服务。

## 7. 外部验收与恢复

[state.py](../../integrated_agent/state.py)读取实际 Task、记忆索引、cron 文件、workflow 快照与 journal，核对已释放 worktree 和保留分支；[verify.py](../../integrated_agent/verify.py)另在受限副本里验证业务。模型调用数、测试数或 workflow completed 均不替代这两层。

workflow 中，`ExecutionState.agent` 先查 schema/预算和语义键缓存；新响应不符合 schema 时只重试一次，两次实际用量合计。`parallel` 等所有已启动分支结束才抛错误，让已成功分支写完 journal。已有的真实跨进程恢复复用了两个成功检查，仅新增一次验证；日志损坏时保留原文件并报错。

门禁用 `critical_gate` 对每个关键条件做 `is True` 判断。一个权限失败或未评分项会阻止接受，不能与其他高分平均。完整17项机制与各自真实/夹具证据边界见 [能力说明](../../integrated_agent/capabilities.md)。

"""Score a saved actual native trace with both pinned metric implementations."""

import argparse
import json
from pathlib import Path

from lessons.e05_faithfulness.eval import evaluate
from lessons.e10_tool_evaluation.metrics import score_rules

from .code import sha, trace_digest, trace_samples


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--input", type=Path, required=True)
    parser.add_argument("--report", type=Path, required=True)
    args = parser.parse_args()
    source = args.input.read_text()
    saved = json.loads(source)
    samples = trace_samples(saved)
    output = {
        "lesson": "e16",
        "origin": "new_native_metric_scoring_of_a_saved_live_integration_trace",
        "source_report_sha256": sha(source),
        "native_trace_sha256": trace_digest(saved),
        "samples": samples,
        "tool_results": [],
        "faithfulness_results": [],
        "agent_new_requests": 0,
        "judge_model_id": None,
        "cost": None,
        "status": "started",
        "judge_requests": 0,
        "metric_errors": 0,
    }
    args.report.parent.mkdir(parents=True, exist_ok=True)
    with args.report.open("x") as handle:
        handle.write(json.dumps(output, ensure_ascii=False, indent=2) + "\n")

    def save():
        temporary = args.report.with_suffix(".tmp")
        temporary.write_text(
            json.dumps(output, ensure_ascii=False, indent=2, allow_nan=False) + "\n"
        )
        temporary.replace(args.report)

    try:
        for sample in samples["tools"]:
            output["tool_results"].append(score_rules(sample))
            save()
        for sample in samples["faithfulness"]:
            result = evaluate(sample)
            output["faithfulness_results"].append(result)
            output["judge_requests"] += len(result["calls"])
            output["metric_errors"] += sum(
                v.get("status") == "error" for v in result["metrics"].values()
            )
            save()
        output["status"] = "returned"
    except Exception as error:  # noqa: BLE001 - save partial scores, never fabricate zero or pass
        output["status"] = "error"
        output["error"] = {"type": type(error).__name__, "message": str(error)}
    finally:
        save()
    print(
        json.dumps(
            {
                "report": str(args.report),
                "status": output["status"],
                "judge_requests": output["judge_requests"],
                "metric_errors": output["metric_errors"],
            }
        )
    )
    return int(output["status"] != "returned" or output["metric_errors"] != 0)


if __name__ == "__main__":
    raise SystemExit(main())
